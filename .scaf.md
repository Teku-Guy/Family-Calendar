# Family Calendar — Scaffold (Next.js + Supabase + Google Calendar Sync)

This repo skeleton gives you:
- Next.js (App Router) + TypeScript + Tailwind
- Supabase schema + minimal RLS
- Google OAuth + Calendar read/write
- Polling sync with `syncToken` (upgrade to push later)
- Web calendar views (`/calendar`, `/kiosk`)
- Pi kiosk + screensaver scripts

---

- Node.js ≥ 20

## 5) Supabase schema & RLS
**supabase/schema.sql**
```sql
create extension if not exists pgcrypto;

create table families (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  owner_id uuid not null,
  created_at timestamptz default now()
);

create type family_role as enum ('owner','parent','teen','kid','guest');

create table family_members (
  family_id uuid references families(id) on delete cascade,
  user_id uuid references auth.users(id) on delete cascade,
  role family_role not null default 'parent',
  color text default '#6366F1',
  primary key (family_id, user_id)
);

create table calendars (
  id uuid primary key default gen_random_uuid(),
  family_id uuid references families(id) on delete cascade,
  name text not null,
  color text default '#10B981',
  is_primary boolean default false
);

create table events (
  id uuid primary key default gen_random_uuid(),
  calendar_id uuid references calendars(id) on delete cascade,
  title text not null,
  description text,
  location text,
  starts_at timestamptz not null,
  ends_at timestamptz not null,
  all_day boolean default false,
  recurrence_rule text,
  event_tz text default 'America/Los_Angeles',
  created_by uuid references auth.users(id),
  updated_at timestamptz default now()
);

create table external_calendars (
  id uuid primary key default gen_random_uuid(),
  family_id uuid references families(id) on delete cascade,
  provider text check (provider in ('google')) not null,
  provider_calendar_id text not null,
  is_primary boolean default true,
  sync_token text
);

create table external_event_links (
  event_id uuid references events(id) on delete cascade,
  provider text not null,
  provider_event_id text not null,
  primary key (event_id, provider)
);
```

**supabase/rls.sql** (turn RLS ON for all tables first)
```sql
alter table families enable row level security;
alter table family_members enable row level security;
alter table calendars enable row level security;
alter table events enable row level security;

-- Helper: is member of family
create or replace function is_member(fam uuid) returns boolean as $$
  select exists(
    select 1 from family_members m
    where m.family_id = fam and m.user_id = auth.uid()
  );
$$ language sql stable;

-- families
create policy fam_read on families for select using (is_member(id));
create policy fam_write on families for all using (
  exists(select 1 from family_members m where m.family_id = id and m.user_id = auth.uid() and m.role in ('owner','parent'))
);

-- calendars
create policy cal_read on calendars for select using (is_member(family_id));
create policy cal_write on calendars for all using (
  exists(select 1 from family_members m where m.family_id = calendars.family_id and m.user_id = auth.uid() and m.role in ('owner','parent'))
);

-- events
create policy evt_read on events for select using (
  exists(select 1 from calendars c join family_members m on m.family_id = c.family_id and m.user_id = auth.uid() where c.id = events.calendar_id)
);
create policy evt_write on events for insert with check (
  exists(select 1 from calendars c join family_members m on m.family_id = c.family_id and m.user_id = auth.uid() where c.id = events.calendar_id and m.role in ('owner','parent','teen'))
);
create policy evt_update on events for update using (
  exists(select 1 from calendars c join family_members m on m.family_id = c.family_id and m.user_id = auth.uid() where c.id = events.calendar_id and m.role in ('owner','parent'))
  or (created_by = auth.uid())
);
create policy evt_delete on events for delete using (
  exists(select 1 from calendars c join family_members m on m.family_id = c.family_id and m.user_id = auth.uid() where c.id = events.calendar_id and m.role in ('owner','parent'))
);
```

## 7) Google API helper
**src/lib/google.ts**
```ts
import { google } from "googleapis";

type Tokens = { access_token: string; refresh_token?: string; expiry_date?: number };

export function googleCalendarClient(tokens: Tokens) {
  const oauth2Client = new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID!,
    process.env.GOOGLE_CLIENT_SECRET!,
    `${process.env.NEXT_PUBLIC_APP_URL}/api/oauth/google/callback`
  );
  oauth2Client.setCredentials(tokens);
  return google.calendar({ version: "v3", auth: oauth2Client });
}
```

## 8) OAuth routes (start + callback)
**src/app/api/oauth/google/start/route.ts**
```ts
import { NextResponse } from "next/server";
import { google } from "googleapis";

export async function GET() {
  const oauth2Client = new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID!,
    process.env.GOOGLE_CLIENT_SECRET!,
    `${process.env.NEXT_PUBLIC_APP_URL}/api/oauth/google/callback`
  );
  const url = oauth2Client.generateAuthUrl({
    access_type: "offline",
    prompt: "consent",
    scope: [
      "https://www.googleapis.com/auth/calendar",
      "https://www.googleapis.com/auth/calendar.events",
      "openid","email","profile"
    ],
  });
  return NextResponse.redirect(url);
}
```

**src/app/api/oauth/google/callback/route.ts**
```ts
import { NextResponse } from "next/server";
import { google } from "googleapis";
import { supabaseAdmin } from "@/lib/supabaseAdmin";

export async function GET(req: Request) {
  const url = new URL(req.url);
  const code = url.searchParams.get("code");
  if (!code) return NextResponse.json({ error: "no_code" }, { status: 400 });

  const oauth2Client = new google.auth.OAuth2(
    process.env.GOOGLE_CLIENT_ID!,
    process.env.GOOGLE_CLIENT_SECRET!,
    `${process.env.NEXT_PUBLIC_APP_URL}/api/oauth/google/callback`
  );
  const { tokens } = await oauth2Client.getToken(code);

  // TODO: identify current user (via your session); for scaffold store to a demo user
  const {
    data: { user },
    error: uerr,
  } = await supabaseAdmin.auth.getUser();
  if (uerr || !user) return NextResponse.json({ error: "auth" }, { status: 401 });

  // Save tokens into a simple table (you can create a dedicated table later)
  await supabaseAdmin.from("profiles").upsert({ user_id: user.id });
  await supabaseAdmin
    .from("external_calendars")
    .upsert({ family_id: null, provider: "google", provider_calendar_id: "primary" })
    .select();

  return NextResponse.redirect(`${process.env.NEXT_PUBLIC_APP_URL}/calendar`);
}
```
> ⚠️ *Note:* In a real app you’ll store per-user tokens in a secure table (not shown here to keep the scaffold compact). Replace the upserts with your own `external_accounts` table.

## 9) Events API (create → DB → Google)
**src/app/api/events/route.ts**
```ts
import { NextResponse } from "next/server";
import { z } from "zod";
import { supabaseAdmin } from "@/lib/supabaseAdmin";
import { googleCalendarClient } from "@/lib/google";

const EventSchema = z.object({
  calendarId: z.string().uuid(),
  title: z.string().min(1),
  description: z.string().optional(),
  location: z.string().optional(),
  startsAt: z.string(),
  endsAt: z.string(),
  allDay: z.boolean().optional(),
  rrule: z.string().optional(),
});

export async function POST(req: Request) {
  const body = EventSchema.parse(await req.json());

  const { data: evt, error } = await supabaseAdmin
    .from("events")
    .insert({
      calendar_id: body.calendarId,
      title: body.title,
      description: body.description,
      location: body.location,
      starts_at: body.startsAt,
      ends_at: body.endsAt,
      all_day: body.allDay ?? false,
      recurrence_rule: body.rrule,
    })
    .select()
    .single();
  if (error) return NextResponse.json({ error: error.message }, { status: 400 });

  // Pseudo: load user tokens -> create google event too
  // const cal = googleCalendarClient(tokens)
  // await cal.events.insert({ calendarId: "primary", requestBody: { summary: body.title, ... } });

  return NextResponse.json(evt);
}
```

import { NextResponse } from "next/server";
import { supabaseAdmin } from "@/lib/supabaseAdmin";

export async function GET() {
  const { data } = await supabaseAdmin
    .from("events")
    .select("id,title,starts_at,ends_at")
    .gte("starts_at", new Date().toISOString())

## 12) Calendar views
**src/components/Calendar.tsx**
```tsx
"use client";
import { useEffect, useMemo, useState } from "react";
import { addDays, startOfWeek } from "date-fns";

export type Evt = { id: string; title: string; starts_at: string; ends_at: string };

export default function Calendar({ initialEvents = [] as Evt[] }) {
  const [events] = useState<Evt[]>(initialEvents);
  const start = startOfWeek(new Date(), { weekStartsOn: 0 });
  const days = Array.from({ length: 7 }, (_, i) => addDays(start, i));
  return (
    <div className="grid grid-cols-7 gap-2 p-4">
      {days.map((d, i) => (
        <div key={i} className="min-h-[60vh] rounded-2xl border border-neutral-800 p-2">
          <div className="text-neutral-400 text-sm mb-2">{d.toDateString()}</div>
          <div className="space-y-2">
            {events.filter(e => new Date(e.starts_at).toDateString() === d.toDateString()).map(e => (
              <div key={e.id} className="rounded-xl border border-neutral-700 px-3 py-2">
                <div className="text-sm font-medium">{e.title}</div>
              </div>
            ))}
          </div>
        </div>
      ))}
    </div>
  );
}
```

**kiosk/slideshow.py**
```python
import os, random, time
from PIL import Image
from PIL import ImageTk
import tkinter as tk
IMG_DIR = os.environ.get("IMG_DIR", "/home/pi/Pictures")
INTERVAL = int(os.environ.get("INTERVAL", "30"))
root = tk.Tk()
root.attributes('-fullscreen', True)
label = tk.Label(root, bg='black')
label.pack(fill='both', expand=True)

def tick():
    files = [os.path.join(IMG_DIR,f) for f in os.listdir(IMG_DIR) if f.lower().endswith((".jpg",".jpeg",".png"))]
    if not files:
        root.after(INTERVAL*1000, tick); return
    img = Image.open(random.choice(files))
    w, h = root.winfo_screenwidth(), root.winfo_screenheight()
    img = img.resize((w,h))
    photo = ImageTk.PhotoImage(img)
    label.config(image=photo)
    label.image = photo
    root.after(INTERVAL*1000, tick)

tick(); root.mainloop()
```
